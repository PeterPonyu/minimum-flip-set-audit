%% Numbered environments, shared by every binding of this paper for the same
%% reason the body is: a statement that is Lemma 2 in one binding and Lemma 3
%% in another cannot be cross-referenced by a reader holding the other one.
\theoremstyle{plain}
\newtheorem{proposition}{Proposition}
\newtheorem{lemma}{Lemma}
\newtheorem{corollary}{Corollary}

\theoremstyle{definition}
\newtheorem{definition}{Definition}
